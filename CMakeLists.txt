cmake_minimum_required(VERSION 3.24)
project(SynapticGenesis LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES native)
endif()
find_package(CUDAToolkit REQUIRED)
add_executable(synapticgenesis src/spike_lm.cu)
target_link_libraries(synapticgenesis PRIVATE CUDA::cudart CUDA::cublas)
target_compile_definitions(synapticgenesis PRIVATE NOMINMAX)
if(MSVC)
  target_compile_options(synapticgenesis PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/W4,/utf-8>)
endif()
enable_testing()
add_test(NAME spiking-native COMMAND synapticgenesis self-test --out ${CMAKE_BINARY_DIR}/test-results)
add_test(NAME spiking-live COMMAND synapticgenesis live-test --out ${CMAKE_BINARY_DIR}/live-test-results)
add_test(NAME spiking-graph COMMAND synapticgenesis graph-test --out ${CMAKE_BINARY_DIR}/graph-test-results)
add_test(NAME spiking-replay COMMAND synapticgenesis replay-test --out ${CMAKE_BINARY_DIR}/replay-test-results)
add_test(NAME spiking-adaptive COMMAND synapticgenesis adaptive-test --out ${CMAKE_BINARY_DIR}/adaptive-test-results)
add_test(NAME spiking-synaptic COMMAND synapticgenesis synaptic-test --out ${CMAKE_BINARY_DIR}/synaptic-test-results)
add_test(NAME spiking-evolution COMMAND synapticgenesis evolution-test --out ${CMAKE_BINARY_DIR}/evolution-test-results)
add_test(NAME spiking-curriculum COMMAND synapticgenesis curriculum-test --out ${CMAKE_BINARY_DIR}/curriculum-test-results)
add_test(NAME spiking-feedback COMMAND synapticgenesis feedback-test --out ${CMAKE_BINARY_DIR}/feedback-test-results)
add_test(NAME spiking-trace COMMAND synapticgenesis trace-test --out ${CMAKE_BINARY_DIR}/trace-test-results)
add_test(NAME spiking-gated COMMAND synapticgenesis gated-test --out ${CMAKE_BINARY_DIR}/gated-test-results)
add_test(NAME spiking-selective COMMAND synapticgenesis selective-test --out ${CMAKE_BINARY_DIR}/selective-test-results)
add_test(NAME spiking-associative COMMAND synapticgenesis associative-test --out ${CMAKE_BINARY_DIR}/associative-test-results)
add_test(NAME spiking-association-layout COMMAND synapticgenesis association-layout-bench --out ${CMAKE_BINARY_DIR}/association-layout-results --rounds 3 --iterations 8)
add_test(NAME spiking-stage-replay COMMAND synapticgenesis stage-replay-test --out ${CMAKE_BINARY_DIR}/stage-replay-test-results)
add_test(NAME spiking-reductions COMMAND synapticgenesis reduction-test --out ${CMAKE_BINARY_DIR}/reduction-test-results)

option(SG_BUILD_TRACE_DIAGNOSTIC "Build the read-only learned-neuron trace diagnostic" OFF)
if(SG_BUILD_TRACE_DIAGNOSTIC)
  add_executable(synaptic-trace-diagnostic tests/learned_trace_dump.cu)
  target_link_libraries(synaptic-trace-diagnostic PRIVATE CUDA::cudart CUDA::cublas)
  target_compile_definitions(synaptic-trace-diagnostic PRIVATE NOMINMAX)
  if(MSVC)
    target_compile_options(synaptic-trace-diagnostic PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=/W4,/utf-8>)
  endif()
endif()
